  * Add the form theme to config/packages/twig.yaml:
      twig:
          form_themes:
              - '@NowoPhoneInputBundle/Form/phone_input_widget.html.twig'

  * Run assets:install and include CSS in your layout:
      php bin/console assets:install
      <!-- flag-icons.min.css + phone_input.css from bundles/nowophoneinput/css/ -->
      <!-- phone picker JS is loaded by the form theme (CSP-safe external script) -->
